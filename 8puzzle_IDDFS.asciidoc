def get_neighbors(state):
    neighbors = []

    zero = state.index(0)
    row = zero // 3
    col = zero % 3

    moves = [
        (-1, 0),   # Up
        (1, 0),    # Down
        (0, -1),   # Left
        (0, 1)     # Right
    ]

    for dr, dc in moves:
        new_row = row + dr
        new_col = col + dc

        if 0 <= new_row < 3 and 0 <= new_col < 3:
            new_zero = new_row * 3 + new_col

            new_state = list(state)

            new_state[zero], new_state[new_zero] = (
                new_state[new_zero],
                new_state[zero]
            )

            neighbors.append(tuple(new_state))

    return neighbors


def depth_limited_dfs(state, goal, limit, visited, path):
    # Goal test
    if state == goal:
        return path

    # Depth limit reached
    if limit == 0:
        return None

    visited.add(state)

    for next_state in get_neighbors(state):

        if next_state not in visited:
            result = depth_limited_dfs(
                next_state,
                goal,
                limit - 1,
                visited,
                path + [next_state]
            )

            if result is not None:
                return result

    visited.remove(state)

    return None


def iddfs(start, goal):
    depth = 0

    while True:
        print("Searching at depth:", depth)

        visited = set()

        result = depth_limited_dfs(
            start,
            goal,
            depth,
            visited,
            [start]
        )

        if result is not None:
            return result

        depth += 1


def main():
    inp = tuple(map(int, input(
        "Enter initial state (e.g. 1 2 3 4 5 6 0 7 8): "
    ).split()))

    goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

    solution = iddfs(inp, goal)

    print("\nSolution found!")
    print("Number of moves:", len(solution) - 1)

    for i, state in enumerate(solution):
        print(f"\nStep {i}:")
        print(state[0:3])
        print(state[3:6])
        print(state[6:9])


main()
