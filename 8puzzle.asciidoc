def pop(stack1):
    if len(stack1) == 0:
        return None
    return stack1.pop()


def push(stack1, state):
    if len(stack1) == 9:
        return False
    stack1.append(state)
    return True


def empty(stack1):
    return len(stack1) == 0


def get_neighbors(state):
    """Generate all possible states by moving 0."""
    neighbors = []

    zero = state.index(0)
    row = zero // 3
    col = zero % 3

    moves = [
        (-1, 0),  # up
        (1, 0),   # down
        (0, -1),  # left
        (0, 1)    # right
    ]

    for dr, dc in moves:
        new_row = row + dr
        new_col = col + dc

        if 0 <= new_row < 3 and 0 <= new_col < 3:
            new_index = new_row * 3 + new_col

            new_state = list(state)
            new_state[zero], new_state[new_index] = (
                new_state[new_index],
                new_state[zero]
            )

            neighbors.append(tuple(new_state))

    return neighbors


def main():
    inp = tuple(map(int, input(
        "Enter initial state (e.g. 1 2 3 4 5 6 0 7 8): "
    ).split()))

    goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

    stack1 = []
    visited = set()
    count = 0

    # Put initial state into the stack
    push(stack1, inp)

    while not empty(stack1):

        st = pop(stack1)

        if st in visited:
            continue

        visited.add(st)

        if st == goal:
            print("Solution found!")
            print("State:", st)
            print(f"Solution found in {count} steps")
            return

        count += 1

        # Generate possible next states
        for next_state in get_neighbors(st):
            if next_state not in visited:
                push(stack1, next_state)

    print("No solution found.")


main()
